    <script>
        function initLucide() {
            if (typeof lucide !== 'undefined') lucide.createIcons();
            else setTimeout(initLucide, 100);
        }

        function showTab(tabId) {
            document.querySelectorAll('.tab-content').forEach(tab => tab.classList.remove('active'));
            document.querySelectorAll('.tab-btn').forEach(btn => {
                btn.classList.remove('border-indigo-600', 'text-indigo-600');
                btn.classList.add('border-transparent', 'text-slate-500');
            });
            document.getElementById(tabId).classList.add('active');
            event.currentTarget.classList.remove('border-transparent', 'text-slate-500');
            event.currentTarget.classList.add('border-indigo-600', 'text-indigo-600');
            if(tabId === 'lesson1') drawLab();
        }

        const sun = document.getElementById('sun');
        const sunSlider = document.getElementById('sunSlider');
        const draggables = document.querySelectorAll('.draggable-tag');
        const dropZones = document.querySelectorAll('.drop-zone');
        let score = 0;
        const correctPlacements = new Set();

        sunSlider.oninput = function() {
            const val = parseInt(this.value);
            sun.style.bottom = val + 'px';
            const container = document.getElementById('sunriseContainer');
            if (val < 45) container.style.background = 'linear-gradient(to bottom, #1a2a6c, #b21f1f, #fdbb2d)';
            else if (val < 90) container.style.background = 'linear-gradient(to bottom, #87CEEB, #E0F2F1)';
            else container.style.background = 'linear-gradient(to bottom, #4facfe 0%, #00f2fe 100%)';
        };

        function enableVerifyMode() {
            document.getElementById('dropZoneContainer').classList.remove('hidden-element');
            document.getElementById('tagPanel').classList.remove('hidden-element');
            document.getElementById('scoreDisplay').classList.remove('hidden-element');
            document.getElementById('warmupInstruction').innerHTML = `
                <p>Drag and drop the correct common points tag into the corresponding stage to verify your idea!</p>
            `;
            document.getElementById('btnVerify').classList.add('hidden-element');
            initLucide();
        }

        draggables.forEach(tag => {
            tag.ondragstart = (e) => {
                e.dataTransfer.setData('text/plain', tag.dataset.id);
                tag.style.opacity = '0.5';
            };
            tag.ondragend = () => tag.style.opacity = '1';
        });

        dropZones.forEach(zone => {
            zone.ondragover = (e) => { e.preventDefault(); zone.classList.add('active'); };
            zone.ondragleave = () => zone.classList.remove('active');
            zone.ondrop = (e) => {
                e.preventDefault();
                zone.classList.remove('active');
                const tagId = e.dataTransfer.getData('text/plain');
                const targetId = zone.dataset.target;

                if (tagId === targetId) {
                    if (!correctPlacements.has(tagId)) {
                        score++;
                        correctPlacements.add(tagId);
                        updateScore();
                    }
                    zone.classList.add('bg-green-600', 'border-green-600');
                    zone.innerHTML = `<span class="text-white text-[12px] font-bold">${document.querySelector(`[data-id="${tagId}"]`).innerText}</span>`;
                    document.querySelector(`[data-id="${tagId}"]`).style.visibility = 'hidden';
                } else {
                    zone.classList.add('bg-red-600');
                    setTimeout(() => zone.classList.remove('bg-red-600'), 1000);
                }
            };
        });

        function updateScore() {
            const display = document.getElementById('scoreDisplay');
            display.innerText = `Score: ${score}/3`;
            if (score === 3) display.innerText = "Excellent! 🎉";
        }

        function resetSunrise() {
            score = 0;
            correctPlacements.clear();
            updateScore();
            dropZones.forEach(zone => {
                const step = zone.id.split('-')[1];
                zone.className = "drop-zone w-1/3 h-14 rounded-lg flex flex-col items-center justify-center text-white text-[10px] font-bold text-center p-1";
                zone.innerHTML = `<span>Stage ${step}</span><span class="opacity-50 italic font-normal">Drop Status Here</span>`;
            });
            draggables.forEach(tag => tag.style.visibility = 'visible');
            sunSlider.value = 30;
            sunSlider.oninput();
        }

        const canvas = document.getElementById('geoCanvas');
        const ctx = canvas.getContext('2d');
        const distSlider = document.getElementById('distSlider');
        const distVal = document.getElementById('distVal');
        const statusLabel = document.getElementById('statusLabel');
        const pointsLabel = document.getElementById('pointsLabel');
        const feedback = document.getElementById('compFeedback');
        const statusBox = document.getElementById('statusBox');
        const promptBox = document.getElementById('promptBox');
        const R = 70;
        const centerX = 160;
        const centerY = 150;

        function drawLab() {
            const d = parseInt(distSlider.value);
            distVal.innerText = d;
            ctx.clearRect(0, 0, canvas.width, canvas.height);
            
            ctx.strokeStyle = '#f1f5f9';
            for(let i=0; i<canvas.width; i+=25) {
                ctx.beginPath(); ctx.moveTo(i,0); ctx.lineTo(i,300); ctx.stroke();
            }

            ctx.beginPath(); ctx.arc(centerX, centerY, R, 0, Math.PI * 2);
            ctx.strokeStyle = '#10b981'; ctx.setLineDash([]); ctx.lineWidth = 3; ctx.stroke();
            ctx.beginPath(); ctx.moveTo(centerX, centerY); ctx.lineTo(centerX, centerY - R);
            ctx.strokeStyle = '#10b981'; ctx.lineWidth = 2; ctx.stroke();
            ctx.fillStyle = '#065f46'; ctx.font = 'bold 12px sans-serif'; ctx.fillText(`R = ${R}`, centerX + 5, centerY - R/2);

            const lineX = centerX + d;
            ctx.beginPath(); ctx.moveTo(centerX, centerY); ctx.lineTo(lineX, centerY);
            ctx.strokeStyle = '#4f46e5'; ctx.setLineDash([5, 5]); ctx.stroke();
            ctx.fillText(`d = ${d}`, centerX + d/2 - 15, centerY + 15);
            ctx.beginPath(); ctx.setLineDash([]); ctx.moveTo(lineX, 20); ctx.lineTo(lineX, 280);
            ctx.strokeStyle = '#f43f5e'; ctx.lineWidth = 4; ctx.stroke();
            
            ctx.fillStyle = '#1e293b'; ctx.beginPath(); ctx.arc(centerX, centerY, 4, 0, Math.PI * 2); ctx.fill();
            ctx.fillText("O", centerX - 15, centerY + 5);

            statusBox.classList.add('hidden-element');
            promptBox.classList.remove('hidden');
            feedback.classList.add('hidden');
            document.querySelectorAll('.comparison-btn').forEach(btn => btn.classList.remove('selected'));
        }

        function checkComparison(symbol) {
            const d = parseInt(distSlider.value);
            let correct = false;
            document.querySelectorAll('.comparison-btn').forEach(btn => btn.classList.remove('selected'));
            
            if (symbol === '<' && d < R) correct = true;
            else if (symbol === '=' && d === R) correct = true;
            else if (symbol === '>' && d > R) correct = true;

            event.currentTarget.classList.add('selected');
            feedback.classList.remove('hidden', 'bg-green-100', 'text-green-700', 'bg-red-100', 'text-red-700');
            
            if (correct) {
                feedback.innerText = "Correct! Position revealed.";
                feedback.classList.add('bg-green-100', 'text-green-700');
                statusBox.classList.remove('hidden-element');
                promptBox.classList.add('hidden');
                
                if (d < R) { statusLabel.innerText = "Intersecting"; pointsLabel.innerText = "(2 shared points)"; }
                else if (d === R) { statusLabel.innerText = "Tangent"; pointsLabel.innerText = "(1 shared point)"; }
                else { statusLabel.innerText = "Disjoint / Separate"; pointsLabel.innerText = "(0 shared points)"; }
            } else {
                feedback.innerText = "Incorrect. Try another relation!";
                feedback.classList.add('bg-red-100', 'text-red-700');
                statusBox.classList.add('hidden-element');
                promptBox.classList.remove('hidden');
            }
        }

        distSlider.oninput = drawLab;
        window.onload = () => {
            initLucide();
            sunSlider.oninput();
            drawLab();
        };
    </script>
